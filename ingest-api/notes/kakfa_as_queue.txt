bash:
set QUEUE_BACKEND=kafka
set KAFKA_BROKERS=localhost:9092
set KAFKA_CLIENT_ID=ingest-api
set KAFKA_TOPIC_EVENTS=events.ingest.v1
npm run dev

powershell:
$env:QUEUE_BACKEND="kafka"
$env:KAFKA_BROKERS="localhost:9092"
$env:KAFKA_CLIENT_ID="ingest-api"
$env:KAFKA_TOPIC_EVENTS="events.ingest.v1"
npm run dev

To consume from the topic to verify messages:
docker exec -it kafka `
  /opt/kafka/bin/kafka-console-consumer.sh `
  --bootstrap-server kafka:29092 `
  --topic events.ingest.v1 `
  --from-beginning
